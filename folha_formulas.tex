{\small\noindent Capítulo 28: Campo Magnético}

\formulabox[blue]{Força Magnética sobre uma Carga}{ $\vec{F}_B =  q \vec{v} \times \vec{B} \quad $ (\si{\newton})}{
$\vec{F}_B$ = Força magnética (\si{\newton}) \\
$q$ = Carga (\si{\coulomb}) \\
$\vec{v}$ = Velocidade da carga (\si{\meter\per\second}) \\
$\vec{B}$ = Campo magnético (\si{\tesla})
}

\formulabox[red]{Densidade de Portadores (Efeito Hall)}{$n = \frac{B i}{V l e} \quad $ (\si{\per\meter\cubed})}{
$n$ = Densidade de portadores (\si{\per\meter\cubed}) \\
$B$ = Campo magnético (\si{\tesla}) \\
$i$ = Corrente (\si{\ampere}) \\
$V$ = Tensão (\si{\volt}) \\
$l$ = Largura (\si{\meter}) \\
$e$ = Carga elementar (\si{\coulomb})
}

\formulabox[blue]{Velocidade em Movimento Circular}{$v = \frac{q r B}{m} \quad $ (\si{\meter\per\second})}{
$v$ = Velocidade (\si{\meter\per\second}) \\
$r$ = Raio da trajetória (\si{\meter}) \\
$B$ = Campo magnético (\si{\tesla}) \\
$q$ = Carga (\si{\coulomb}) \\
$m$ = Massa (\si{\kilogram})
}

\formulabox[blue]{Força Magnética em Fio com Corrente}{$\vec{F}_B =  i\vec{L} \times \vec{B} \quad $ (\si{\newton})}{
$\vec{F}_B$ = Força magnética (\si{\newton}) \\
$i$ = Corrente (\si{\ampere}) \\
$\vec{L}$ = Comprimento do fio (\si{\meter}) \\
$\vec{B}$ = Campo magnético (\si{\tesla})
}

\formulabox[teal]{Torque em Espira}{$\vec{\tau} = \vec{\mu} \times \vec{B} \quad $ (\si{\newton\meter})}{
$\vec{\tau}$ = Torque (\si{\newton\meter}) \\
$\vec{\mu}$ = Momento magnético (\si{\ampere\meter\squared}) \\
$\vec{B}$ = Campo magnético (\si{\tesla})
}

\columnbreak

\formulabox[teal]{Energia Potencial Magnética}{$U(\theta) =  -\vec{\mu} \cdot \vec{B} \quad $ (\si{\joule})}{
$U$ = Energia potencial (\si{\joule}) \\
$\vec{\mu}$ = Momento magnético (\si{\ampere\meter\squared}) \\
$\vec{B}$ = Campo magnético (\si{\tesla})
}

\formulabox[blue]{Campo B de um Condutor com Corrente}{$d\vec{B} = \frac{\mu_0}{4\pi} \frac{i d\vec{s}\times\hat{r}}{r^2} \quad $ (\si{\tesla})}{
$d\vec{B}$ = Elemento de campo magnético (\si{\tesla}) \\
$i$ = Corrente (\si{\ampere}) \\
$d\vec{s}$ = Elemento de comprimento (\si{\meter}) \\
$\hat{r}$ = Vetor unitário posição \\
$r$ = Distância (\si{\meter})
}

\formulabox[blue]{Campo B (Biot-Savart)}{$B = \frac{\mu_0 i}{2 \pi R} \quad $ (\si{\tesla})}{
$B$ = Campo magnético (\si{\tesla}) \\
$i$ = Corrente (\si{\ampere}) \\
$R$ = Distância (\si{\meter})
}

\formulabox[teal]{Campo B de um Arco de Circunferência}{$B = \frac{\mu_0 i \theta}{4\pi R} \quad $ (\si{\tesla})}{
$B$ = Campo magnético (\si{\tesla}) \\
$i$ = Corrente (\si{\ampere}) \\
$\theta$ = Ângulo (\si{\radian}) \\
$R$ = Raio (\si{\meter})
}

$i$ = Corrente (\si{\ampere}) \\
