% Exemplo de fórmula de cálculo
\textbf{\int_a^b f(x) dx} \\
Integral definida de $f(x)$ no intervalo $[a, b]$ \\
$a, b$ = limites de integração \\
$f(x)$ = função integrável \\
\vspace{1em}

% Derivada
\textbf{f'(x) = \lim\limits_{h \to 0} \frac{f(x+h) - f(x)}{h}} \\
Definição de derivada \\
$f(x)$ = função \\
h = incremento \\
\vspace{1em}

% Regra do produto
\textbf{(fg)' = f'g + fg'} \\
Derivada do produto de funções \\
